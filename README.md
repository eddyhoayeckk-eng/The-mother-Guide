function phaseInfo(){
 // Approximation based on a known new-moon reference date; not an astronomical ephemeris.
 const synodic=29.53058867;

 const age=((Date.now()-ref)/86400000%synodic+synodic)%synodic;
 let name,emoji,meaning;
 if(age<1.85){name="New Moon";emoji="🌑";meaning="A symbolic time for beginnings, quiet intentions and planning."}
 else if(age<5.53){name="Waxing Crescent";emoji="🌒";meaning="A symbolic time to take a first small step."}
 else if(age<9.22){name="First Quarter";emoji="🌓";meaning="A symbolic time to meet challenges with patience."}
 else if(age<12.91){name="Waxing Gibbous";emoji="🌔";meaning="A symbolic time to refine and keep practising."}
 else if(age<16.61){name="Full Moon";emoji="🌕";meaning="A symbolic time for awareness, gratitude and reflection."}
 else if(age<20.30){name="Waning Gibbous";emoji="🌖";meaning="A symbolic time to share what you have learned."}
 else if(age<23.99){name="Last Quarter";emoji="🌗";meaning="A symbolic time to review and simplify."}
 else if(age<27.68){name="Waning Crescent";emoji="🌘";meaning="A symbolic time for rest and letting go."}

}

function download(filename,text){

 const url=URL.createObjectURL(blob),a=document.createElement("a");
 a.href=url;a.download=filename;a.click();URL.revokeObjectURL(url);
}
